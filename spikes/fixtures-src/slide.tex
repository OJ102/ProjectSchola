\begin{frame}{Gradient Descent in Three Steps}
\begin{itemize}
\item Compute the gradient of the loss with respect to the weights.
\item Step against the gradient: $w \leftarrow w - \eta \nabla L(w)$.
\item Repeat until the validation loss stops improving.
\end{itemize}
\end{frame}
